# VTSearch - All image embedders (CPU)
#
# Image-only deployment that ships every supported image embedder:
#   * SigLIP   (google/siglip-base-patch16-224)         - default, text-capable, ungated
#   * SigLIP 2 (google/siglip2-base-patch16-224)        - text-capable, ungated
#   * CLIP     (openai/clip-vit-base-patch32)           - text-capable, ungated
#   * DINOv2   (facebook/dinov2-base)                   - image-only, ungated
#   * DINOv3   (facebook/dinov3-vitb16-pretrain-lvd1689m) - image-only, gated
#   * EUPE     (facebook/PE-Core-B16-224)               - image-only, gated
#
# Used by docker/Dockerfile.image-embedders. Skips audio / video / text / document
# stacks so we still produce a slim image (only image-side deps are pulled in).

# ── Core framework ───────────────────────────────────────────────────
flask
gunicorn
numpy
requests
tqdm
scikit-learn
# SigLIP 2 + DINOv3 require recent transformers ( >= 4.49 ships SigLIP 2,
# and DINOv3's HF wrapper landed in the same window ). Pin a known-good
# floor instead of "latest" so the image is reproducible.
transformers>=4.49.0
pandas

# ── Eval / visualization ────────────────────────────────────────────
matplotlib
scipy

# ── PyTorch (CPU) ────────────────────────────────────────────────────
--extra-index-url https://download.pytorch.org/whl/cpu
torch>=2.0.0
# DINOv2 / DINOv3 use `AutoImageProcessor`, which transformers loads via
# the torchvision backend (`use_fast=True` default in recent releases).
# Without torchvision the processor import raises ImportError at build
# time, so install the matching CPU wheel.
torchvision

# ── Image media type ─────────────────────────────────────────────────
Pillow

# ── SigLIP / SigLIP 2 / CLIP tokenizers ──────────────────────────────
sentencepiece
protobuf
# CLIP's tokenizer + SigLIP 2 fast processor both prefer the Rust
# tokenizer crate; install it eagerly so first-load doesn't fall back
# to the slow Python path.
tokenizers

# ── EUPE (Perception Encoder) ────────────────────────────────────────
# EUPE ships custom modeling code on the Hub; loading uses
# trust_remote_code=True. einops is the only extra runtime dep its
# custom code typically pulls in.
einops
