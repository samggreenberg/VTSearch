# VTSearch: LabBench core dependencies (every LabBench media type)
#
# Pared-down requirements for the LabBench deployment: browsing/voting on a
# dataset of one media type with that type's default embedder. This file is
# the part every LabBench image shares; it is not installable on its own as a
# working image. Each supported media type adds its own deps in
# labbench-<type>.txt, which pulls this file in with `-r labbench.txt`:
#
#   labbench-image.txt   image search with SigLIP
#   labbench-audio.txt   audio search with CLAP General
#
# docker/Dockerfile.labbench installs this file first, as a layer every
# MEDIA_TYPE build shares, then requirements/labbench-$MEDIA_TYPE.txt. Video,
# document, text and face deps are skipped to keep the images small.

# ── Core framework ───────────────────────────────────────────────────
flask
flask-smorest
marshmallow
pydantic>=2
werkzeug
gunicorn
numpy
requests
tqdm
scikit-learn
transformers
# One pandas major everywhere; see the pandas entry in pyproject.toml (#4381).
pandas>=3

# ── Eval / visualization ────────────────────────────────────────────
matplotlib
scipy

# ── VTSBrowse projection ─────────────────────────────────────────────
# The Browse canvas fits UMAP on the embedding matrix (vtscore.gpu_backends
# .umap_fit_transform, imported lazily). Needed by every deployment
# regardless of media type: nothing else pulls umap-learn in transitively,
# so omitting it leaves Browse broken at runtime rather than at build time.
umap-learn

# ── PyTorch (CPU) ────────────────────────────────────────────────────
--extra-index-url https://download.pytorch.org/whl/cpu
# Floor: transformers >= 5.18 cannot import a model class on torch 2.6 when no
# GPU is visible (#4838); kept equal to pyproject.toml's.
torch>=2.7
# The CPU wheel index above applies to the whole pip invocation, so a
# labbench-<type>.txt that adds torchvision resolves it against that index.

# ── VTSBrowse signpost naming (toponymy) ─────────────────────────────
# `toponymy` letters the Browse map (vtscore.projection.signpost_build).
# It is NOT listed here: its `transformers<5.0.0` pin would drag the image's
# transformers stack backwards, and the pin is empirically unnecessary for
# our usage — so docker/Dockerfile.labbench installs it in a dedicated
# `--no-deps` step, exactly as scripts/install.sh does for a host install
# (see docs/SETUP.md). A requirements file cannot express
# that: pip rejects `--no-deps` inside one.
#
# What DOES belong here are toponymy's real dependencies, declared on its
# behalf because the `--no-deps` install skips them and this image installs
# the package itself with `--no-deps -e .` (so pyproject.toml's declarations
# never reach it either). Only the ones with no carrier are listed: jinja2
# arrives with flask, numba with fast_hdbscan, and httpx / tokenizers with
# transformers' huggingface_hub. apricot-select ships a legacy setup.py
# sdist, so it is installed alongside toponymy in that same dedicated step.
fast_hdbscan
vectorizers
tenacity
