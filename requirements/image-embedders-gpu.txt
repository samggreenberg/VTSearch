# VTSearch: All image embedders (GPU / CUDA)
#
# GPU twin of requirements/image-embedders.txt. Identical Python deps; the
# only difference is that the CPU PyTorch extra-index-url is omitted so the
# Dockerfile can pass --extra-index-url https://download.pytorch.org/whl/cu121
# at install time. Used by docker/Dockerfile.image-embedders.gpu.

# ── Core framework ───────────────────────────────────────────────────
flask
flask-smorest
marshmallow
pydantic>=2
werkzeug
gunicorn
numpy
requests
tqdm
scikit-learn
# SigLIP 2 + DINOv3 need a recent transformers, and the range needs a
# **ceiling** as well as a floor: transformers v5 moved the plain
# `SiglipImageProcessor` name onto the torchvision implementation and renamed
# the PIL one to `SiglipImageProcessorPil`. That is not a cosmetic rename -- on
# the shipped default embedder the two disagree on **58% of preprocessed
# pixels**, by up to two 8-bit levels (issue #3160, 64 VG images), so an
# unbounded range means two installs of the same commit can embed the same
# image differently. A floor alone is not reproducibility.
transformers>=5.0.0,<6
pandas

# ── Eval / visualization ────────────────────────────────────────────
matplotlib
scipy

# ── VTSBrowse projection ─────────────────────────────────────────────
# The Browse canvas fits UMAP on the embedding matrix (vtscore.gpu_backends
# .umap_fit_transform, imported lazily). Needed by every deployment
# regardless of media type: nothing else pulls umap-learn in transitively,
# so omitting it leaves Browse broken at runtime rather than at build time.
# This image ships no cuML, so the CPU umap-learn path is the only one.
umap-learn

# ── PyTorch (GPU; CUDA version set via --extra-index-url at install time)
torch>=2.0.0

# ── Image media type ─────────────────────────────────────────────────
Pillow

# ── SigLIP / SigLIP 2 / CLIP tokenizers ──────────────────────────────
sentencepiece
protobuf
# CLIP's tokenizer + SigLIP 2 fast processor both prefer the Rust
# tokenizer crate; install it eagerly so first-load doesn't fall back
# to the slow Python path.
tokenizers

# ── EUPE (Perception Encoder) ────────────────────────────────────────
# EUPE ships custom modeling code on the Hub; loading uses
# trust_remote_code=True. einops is the only extra runtime dep its
# custom code typically pulls in.
einops

# ── SigLIP-L (open_clip ViT-SO400M-14-SigLIP-384) ────────────────────
# Loaded through open_clip (not transformers) so its 1152-d vectors match
# galleries embedded by open_clip itself. timm / ftfy are open_clip's own
# runtime deps.
open_clip_torch
timm
ftfy
